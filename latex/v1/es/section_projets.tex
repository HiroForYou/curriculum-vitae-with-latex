\sectionTitle{Proyectos}{\faLaptop}
\begin{projects}
    \project{uniMedic}{2020 - 2021}{\github{uniMedic}}{Se desarrolló un chatbot médico usando Deep Learning para diagnósticos en tiempo real. La aplicación móvil se creó con React Native, el backend con Express.js para gestionar registros e historiales médicos, y Flask para consumir modelos de Inteligencia Artificial. Desplegado en la nube de AWS usando Amplify, Cognito y DynamoDB.} {React Native, Express.js, Flask, AWS, Pytorch}

    \project{RasPi-BloodView}{2020 - 2021}{\github{Enigma-A-I/RasPi-BloodView}} {Sistema de detección de células sanguíneas desplegado en una Raspberry Pi 4 usando PyTorch. El modelo de red neuronal convolucional permite un reconocimiento rápido y económico de 3 tipos de células sanguíneas. Cuenta con una interfaz personalizada en HTML/CSS renderizada con Flask y un sistema de gestión de archivos para almacenamiento seguro en el servidor.} {PyTorch, Raspberry Pi}
\end{projects}
